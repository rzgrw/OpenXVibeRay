cmake_minimum_required(VERSION 3.20)
project(xrSimStandalone CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
file(GLOB XRSIM_SRC "${ROOT}/src/xrEngine/xrSim/*.cpp")
add_library(xrSimCore STATIC ${XRSIM_SRC})
target_include_directories(xrSimCore PUBLIC "${ROOT}/src/xrEngine" "${ROOT}/src")
find_package(CURL)
if(CURL_FOUND)
  target_compile_definitions(xrSimCore PRIVATE XRAY_AGENT_HTTP_CURL=1)
  target_link_libraries(xrSimCore PUBLIC CURL::libcurl)
endif()
foreach(t xrSimWorldStateTests xrSimAuthoritySpineTests)
  add_executable(${t} "${ROOT}/src/xrEngine/xrSim/tests/${t}.cpp")
  target_link_libraries(${t} PRIVATE xrSimCore)
endforeach()
